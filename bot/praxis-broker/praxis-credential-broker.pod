# The inference broker's pod: Praxis (praxis-credential-broker-praxis) and
# the pod's own tailnet node (praxis-credential-broker-tailscale), which
# share its network. Set up as README.md says, then:
#   ln -s $PWD/bot/praxis-broker ~/.config/containers/systemd/
#   systemctl --user daemon-reload && systemctl --user start praxis-credential-broker-pod
# Names are upstream's (cgwalters-bot/praxis-credential-broker's
# contrib/quadlet), so its scripts and INTERNALS.md apply as they are.
[Unit]
Description=Praxis credential broker pod: Claude for agentic-job runs, on the tailnet
StartLimitIntervalSec=0
Wants=network-online.target
After=network-online.target

[Pod]
PodName=praxis-credential-broker
ExitPolicy=continue
# Praxis listens on 8081 in the pod. This machine reaches it on its own
# loopback only, for the checks in README.md; the tailnet reaches it
# through the pod's tailnet node, on that node's address.
PublishPort=127.0.0.1:18080:8081

[Service]
Restart=on-failure
RestartSec=15s
TimeoutStartSec=120s

[Install]
WantedBy=default.target
